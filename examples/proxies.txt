# Example proxy list for Tor's Hammer 2.0
# ========================================
# One proxy per line. Blank lines and lines starting with # are ignored.
#
# Supported proxy schemes:
#   socks5://[user:pass@]host:port     - SOCKS5 with optional authentication
#   socks4://[user@]host:port          - SOCKS4a (supports remote DNS)
#   http://[user:pass@]host:port       - HTTP CONNECT proxy
#
# Examples for different proxy types:
# ===================================

# Tor (default SOCKS5 port)
socks5://127.0.0.1:9050

# HTTP proxy (no authentication)
http://proxy.example.com:8080
http://192.168.1.100:3128

# SOCKS5 with authentication
socks5://username:password@proxy.example.com:1080
socks5://user:secret@10.0.0.42:1080

# SOCKS4a proxy
socks4://proxy.example.com:1080
socks4://192.168.1.50:1080

# HTTP proxy with authentication
http://user:pass@proxy.example.com:8080
http://admin:secureproxy@192.168.1.200:3128

# Multiple proxies for rotation (use with --rotate-proxies flag)
# The tool will randomly select one proxy per connection
socks5://proxy1.example.com:1080
socks5://proxy2.example.com:1080
http://proxy3.example.com:8080
socks4://proxy4.example.com:1080
